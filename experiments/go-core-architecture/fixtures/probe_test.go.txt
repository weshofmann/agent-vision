func start(t *testing.T, script string) (*exec.Cmd, *os.File, int, <-chan error) {
 t.Helper()
 c := exec.Command("/bin/sh", "-c", script)
 c.Env = []string{"PATH=/usr/bin:/bin", "TERM=xterm-256color", "HOME=/nonexistent"}
 f,e := pty.StartWithSize(c, &pty.Winsize{Rows:24, Cols:80}); if e!=nil {t.Fatal(e)}
 fd:=int(f.Fd()); if e=syscall.SetNonblock(fd,true);e!=nil {t.Fatal(e)}
 done:=make(chan error,1); go func(){done<-c.Wait()}()
 t.Cleanup(func(){f.Close(); c.Process.Kill()})
 return c,f,fd,done
}
func TestInteractiveShellAndForegroundGroup(t *testing.T){

 f,e:=pty.StartWithSize(c,&pty.Winsize{Rows:24,Cols:80});if e!=nil{t.Fatal(e)}

 done:=make(chan error,1);go func(){done<-c.Wait()}()

 if errno!=0||int(fg)!=c.Process.Pid{t.Fatal("foreground pgrp")}
 syscall.Write(fd,[]byte("stty -echo; case $- in *i*) printf 'MODE_%s\\n' INTERACTIVE;; *) printf NONINTERACTIVE;; esac\n"))
 readUntil(t,fd,[]byte("MODE_INTERACTIVE"))

 // Keep draining while the kernel/shell finishes, rather than Wait-before-read.
 end:=time.Now().Add(2*time.Second);finished:=false;b:=make([]byte,4096)
 for time.Now().Before(end)&&!finished{select{case <-done:finished=true;default:syscall.Read(fd,b);time.Sleep(time.Millisecond)}}
 if !finished{f.Close();c.Process.Kill();t.Fatal("interactive drain/wait bound")}
 s:=c.ProcessState.Sys().(syscall.WaitStatus);f.Close();if !s.Exited()||s.ExitStatus()!=0{t.Fatal("interactive exit")}
 t.Log("no-argument sh detects interactive tty=true foreground-group equals direct shell=true")
}
func readUntil(t *testing.T, fd int, marker []byte) []byte {
 t.Helper(); var out []byte; b:=make([]byte,4096); end:=time.Now().Add(2*time.Second)
 for time.Now().Before(end) {n,e:=syscall.Read(fd,b); if n>0 {out=append(out,b[:n]...)}; if bytes.Contains(out,marker){return out}; if e!=nil && e!=syscall.EAGAIN && e!=syscall.EINTR {t.Fatalf("read: %v",e)}; time.Sleep(time.Millisecond)}
 t.Fatalf("marker missing, bytes=%d",len(out));return nil
}
func wait(t *testing.T, c *exec.Cmd, done <-chan error) syscall.WaitStatus {
 t.Helper();select {case <-done:case <-time.After(2*time.Second):t.Fatal("wait bound")};return c.ProcessState.Sys().(syscall.WaitStatus)
}

func TestSerializedReaper(t *testing.T){
 // Prototype the macOS safe ownership alternative: only this goroutine reaps/signals.
 for i:=0;i<50;i++{
 c:=exec.Command("/bin/sh","-c","stty -echo; printf READY; read x; exit 17")
 c.Env=[]string{"PATH=/usr/bin:/bin","HOME=/nonexistent"}
 f,e:=pty.StartWithSize(c,&pty.Winsize{Rows:24,Cols:80});if e!=nil{t.Fatal(e)}
 fd:=int(f.Fd());syscall.SetNonblock(fd,true);readUntil(t,fd,[]byte("READY"))
 if i%2==0 {syscall.Write(fd,[]byte("done\n"))} else {f.Close()}
 var status syscall.WaitStatus;reaped:=false;signalled:=false;begun:=time.Now()
 for time.Since(begun)<time.Second {
 n,e:=syscall.Wait4(c.Process.Pid,&status,syscall.WNOHANG,nil)
 if n==c.Process.Pid {reaped=true;break};if e!=nil&&e!=syscall.EINTR{t.Fatal(e)}
 if !signalled && time.Since(begun)>250*time.Millisecond {if e=syscall.Kill(c.Process.Pid,syscall.SIGKILL);e!=nil&&e!=syscall.ESRCH{t.Fatal(e)};signalled=true}
 time.Sleep(time.Millisecond)
 }
 if !reaped{t.Fatal("serialized reaper bound")};f.Close();c.Process.Release()
 if i%2==0 && (!status.Exited()||status.ExitStatus()!=17){t.Fatal("natural status")}
 }
 t.Log("50 natural/close cycles; one Wait4/signalling owner; exact natural exit17; Process.Release after reaping")
}
